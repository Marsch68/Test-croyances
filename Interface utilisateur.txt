const questionsBase = [
  {
    id: "q_exist",
    texte: "L'être humain naît sans nature prédéfinie ; par nos choix libres, nous créons le sens de notre existence dans un univers indifférent.",
    scores: {
      "Existentialisme": 5,
      "Catholicisme": -4,
      "Spinozisme": -5,
      "Agnosticisme": 2
    }
  },
  {
    id: "q_dogme",
    texte: "La vérité provient d'une révélation (textes sacrés, prophètes). Elle est immuable et les textes doivent être lus au premier degré.",
    scores: {
      "Salafisme": 5,
      "Catholicisme": 3,
      "Déisme": -5,
      "Existentialisme": -5
    }
  },
  {
    id: "q_spinoza",
    texte: "Dieu n'est pas une personne, c'est l'ordre absolu, nécessaire et géométrique de la Nature.",
    scores: {
      "Spinozisme": 5,
      "Déisme": 2,
      "Catholicisme": -5,
      "Salafisme": -5
    }
  }
];

// Variables d'état
let indexCourant = 0;
const reponsesUtilisateur = {}; 

const elSondage = document.getElementById("sondage");
const elResultats = document.getElementById("resultats");
const elQuestionTexte = document.getElementById("question-texte");
const elProgress = document.getElementById("progress-text");

function afficherQuestion() {
  const q = questionsBase[indexCourant];
  elQuestionTexte.textContent = q.texte;
  elProgress.textContent = `Question ${indexCourant + 1} / ${questionsBase.length}`;
}

function enregistrerChoix(valeur) {
  const qId = questionsBase[indexCourant].id;
  reponsesUtilisateur[qId] = valeur;
  indexCourant++;

  if (indexCourant < questionsBase.length) {
    afficherQuestion();
  } else {
    terminerSondage();
  }
}

// --- LE MOTEUR DE CALCUL ---
function calculerAffinities(reponses) {
  const totaux = {};
  const maxPossibles = {};

  // Étape A : Parcourir les réponses et cumuler les points
  for (const [qId, valeurChoisie] of Object.entries(reponses)) {
    const question = questionsBase.find(q => q.id === qId);
    
    if (question && question.scores) {
      for (const [profil, poids] of Object.entries(question.scores)) {
        // Initialiser les compteurs si le profil est rencontré pour la première fois
        if (!totaux[profil]) {
          totaux[profil] = 0;
          maxPossibles[profil] = 0;
        }

        // Le calcul clé :
        // Si le poids est négatif (ex: -5) et qu'on répond "Pas d'accord" (-2), 
        // les signes s'annulent et on gagne 10 points pour ce profil !
        totaux[profil] += poids * valeurChoisie;
        
        // Le max possible pour cette question (valeur absolue du poids * 2)
        maxPossibles[profil] += Math.abs(poids) * 2;
      }
    }
  }

  // Étape B : Convertir en pourcentages
  const resultats = [];
  for (const profil in totaux) {
    if (maxPossibles[profil] > 0) {
      // (Total / Max) * 100. On utilise Math.max(0, ...) pour ne pas afficher de scores en dessous de 0%.
      let pourcentage = Math.max(0, (totaux[profil] / maxPossibles[profil]) * 100);
      resultats.push({ 
        profil: profil, 
        pourcentage: Math.round(pourcentage) 
      });
    }
  }

  // Trier du plus grand pourcentage au plus petit
  resultats.sort((a, b) => b.pourcentage - a.pourcentage);
  return resultats;
}

// --- AFFICHAGE DES RÉSULTATS ---
function terminerSondage() {
  elSondage.style.display = "none";
  elResultats.style.display = "block";
  
  const affinites = calculerAffinities(reponsesUtilisateur);
  
  // Vider le conteneur de résultats
  elResultats.innerHTML = "<h2>Votre profil philosophique</h2><p>Voici les systèmes de pensée dont vous êtes le plus proche :</p>";
  
  // Générer les barres HTML pour chaque profil
  affinites.forEach(aff => {
    // On ne montre que les profils au-dessus de 0% pour ne pas polluer l'affichage
    if (aff.pourcentage > 0) {
      const htmlCode = `
        <div class="resultat-item">
          <div class="resultat-header">
            <span>${aff.profil}</span>
            <span>${aff.pourcentage}%</span>
          </div>
          <div class="barre-fond">
            <div class="barre-remplissage" style="width: ${aff.pourcentage}%;"></div>
          </div>
        </div>
      `;
      elResultats.innerHTML += htmlCode;
    }
  });
}
