  <script>
    /* ----- JAVASCRIPT : Moteur logique ----- */

    // Simulation de votre futur fichier JSON
    const questionsBase = [
      {
        id: "q_existentialisme",
        texte: "L'être humain naît sans nature prédéfinie ; par nos choix libres, nous créons le sens de notre existence dans un univers indifférent."
      },
      {
        id: "q_dogme_litteral",
        texte: "La vérité provient d'une révélation (textes sacrés, prophètes). Elle est immuable et les textes doivent être lus au premier degré."
      },
      {
        id: "q_spinoza",
        texte: "Dieu n'est pas une personne, c'est l'ordre absolu et géométrique de la Nature."
      }
    ];

    // Variables d'état
    let indexCourant = 0;
    const reponsesUtilisateur = {}; // Stockera { "id_question": score }

    // Éléments du DOM
    const elSondage = document.getElementById("sondage");
    const elResultats = document.getElementById("resultats");
    const elQuestionTexte = document.getElementById("question-texte");
    const elProgress = document.getElementById("progress-text");
    const elDebug = document.getElementById("debug-reponses");

    // Fonction pour afficher la question courante
    function afficherQuestion() {
      const q = questionsBase[indexCourant];
      elQuestionTexte.textContent = q.texte;
      elProgress.textContent = `Question ${indexCourant + 1} / ${questionsBase.length}`;
    }

    // Fonction déclenchée au clic sur un bouton
    function enregistrerChoix(valeur) {
      const qId = questionsBase[indexCourant].id;
      
      // Enregistre la réponse dans le dictionnaire d'état
      reponsesUtilisateur[qId] = valeur;

      // Passe à la suite
      indexCourant++;

      if (indexCourant < questionsBase.length) {
        afficherQuestion();
      } else {
        terminerSondage();
      }
    }

    // Fonction de fin de test
    function terminerSondage() {
      elSondage.style.display = "none";
      elResultats.style.display = "block";
      
      // Pour l'instant, on affiche simplement l'objet JS pour vérifier que ça marche
      elDebug.textContent = JSON.stringify(reponsesUtilisateur, null, 2);
      
      // C'est ici que vous appellerez vos fonctions :
      // calculerAffinities(reponsesUtilisateur);
      // genererBilanDissonance(reponsesUtilisateur);
    }

    // Initialisation au chargement de la page
    window.onload = afficherQuestion;

  </script>
